{{/* Usage : {{< figure src="image.jpg" alt="…" caption="…" width="170" class="aligncenter" >}} src = ressource du page bundle ; image redimensionnée en WebP, alt toujours présent. */}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default "" -}} {{- $w := .Get "width" -}} {{- $class := .Get "class" -}} {{- $cap := .Get "caption" -}} {{- with .Page.Resources.Get $src -}} {{- $max := cond (ne $w "") (mul (int $w) 2) 1600 -}} {{- $max = cond (gt $max .Width) .Width $max -}} {{- $img := .Resize (printf "%dx webp" $max) -}} {{- with $cap }}{{ end }} {{- else -}} {{- errorf "figure : ressource %q introuvable dans %s" $src $.Page.File.Path -}} {{- end -}}
